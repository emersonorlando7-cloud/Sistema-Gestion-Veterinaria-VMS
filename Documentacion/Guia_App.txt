# Decorador para verificar que el usuario haya iniciado sesión.
def login_required(f):
    @wraps(f)
    def decorated_function(*args, **kwargs):
        # Si no hay 'user_id' en la sesión, significa que no está logueado.
        if 'user_id' not in session:
            # Muestra un mensaje de advertencia.
            flash('Por favor, inicia sesión primero.', 'warning')
            # Redirige al login.
            return redirect(url_for('login'))
        # Si está logueado, ejecuta la función original.
        return f(*args, **kwargs)
    return decorated_function

# Decorador para verificar que el rol del usuario tenga un permiso específico.
def permission_required(permission):
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            # Si no está logueado, lo envía al login.
            if 'user_id' not in session:
                return redirect(url_for('login'))
            # Obtiene el rol del usuario de la sesión.
            user_role = session.get('role')
            # Los roles 'admin' y 'sistemas' tienen acceso total a todo.
            if user_role in ['admin', 'sistemas']:
                return f(*args, **kwargs)
            # Busca el rol en la base de datos.
            role_db = Role.query.filter_by(name=user_role).first()
            # Si el rol tiene el permiso o el permiso 'all', permite el acceso.
            if role_db and (permission in role_db.permissions.split(',') or 'all' in role_db.permissions.split(',')):
                return f(*args, **kwargs)
            # Si no tiene permiso, muestra un mensaje flash de peligro y redirige al inicio.
            flash('⛔ No tienes permisos para acceder a esta sección.', 'danger')
            return redirect(url_for('inicio'))
        return decorated_function
    return decorator

# Función auxiliar para verificar si el usuario puede agregar notas a la bitácora.
def _puede_agregar_nota():
    # Obtiene el rol de la sesión.
    user_role = session.get('role')
    # Roles con permiso por defecto para escribir notas.
    if user_role in ['admin', 'sistemas', 'veterinario', 'recepcionista']:
        return True
    # Si es un rol personalizado, verifica sus permisos.
    if user_role:
        # Busca el rol en la base de datos.
        role_db = Role.query.filter_by(name=user_role).first()
        if role_db:
            # Divide la cadena de permisos por comas.
            perms = role_db.permissions.split(',')
            # Si tiene permisos de creación/edición o 'all', puede escribir notas.
            if 'editar_pacientes' in perms or 'crear_pacientes' in perms or 'all' in perms:
                return True
    # Si no cumple ninguna condición, no puede escribir.
    return False

# --- RUTA: LOGIN ---
@app.route('/login', methods=['GET', 'POST'])
def login():
    # Si el usuario envía el formulario (POST).
    if request.method == 'POST':
        # Captura el nombre de usuario del formulario.
        username = request.form['username']
        # Captura la contraseña del formulario.
        password = request.form['password']
        try:
            # Busca al usuario en la base de datos.
            user = User.query.filter_by(username=username).first()
            # Verifica que el usuario exista y que la contraseña coincida con el hash.
            if user and check_password_hash(user.password_hash, password):
                # Guarda el ID del usuario en la sesión.
                session['user_id'] = user.id
                # Guarda el nombre del usuario en la sesión.
                session['username'] = user.username
                # Guarda el rol del usuario en la sesión.
                session['role'] = user.role
                # Redirige al panel de inicio.
                return redirect(url_for('inicio'))
            else:
                # Si las credenciales son incorrectas, muestra un error.
                flash('Usuario o contraseña incorrectos', 'danger')
        except Exception:
            # Si hay un error de conexión a la base de datos, muestra un mensaje.
            flash('Error de conexión a la base de datos.', 'danger')
    # Renderiza la plantilla del formulario de login.
    return render_template('login.html')

# --- RUTA: INICIO (Panel de Control) ---
@app.route('/inicio')
@login_required
def inicio():
    try:
        # Cuenta el total de pacientes registrados.
        total_pacientes = Pet.query.count()
        # Cuenta el total de dueños registrados.
        total_duenos = Owner.query.count()
        # Cuenta el total de consultas registradas.
        total_consultas = Consultation.query.count()

        # Obtiene la fecha actual del sistema.
        hoy = datetime.now().date()
        # Crea un objeto datetime con la fecha de hoy a las 00:00:00.
        inicio_dia = datetime.combine(hoy, dt_time.min)
        # Suma un día para obtener mañana a las 00:00:00.
        fin_dia = inicio_dia + timedelta(days=1)
        
        # Suma 15 días a partir de mañana para limitar las próximas citas.
        fin_rango_futuras = fin_dia + timedelta(days=15)

        # Cuenta las citas programadas entre las 00:00 y las 23:59 de hoy.
        consultas_hoy = Consultation.query.filter(
            Consultation.appointment_date >= inicio_dia,
            Consultation.appointment_date < fin_dia
        ).count()

        # Trae los detalles de las citas de hoy haciendo un JOIN con Mascotas y Dueños.
        citas_hoy = db.session.query(Consultation, Pet, Owner).join(
            Pet, Consultation.pet_id == Pet.id
        ).join(
            Owner, Pet.owner_id == Owner.id
        ).filter(
            Consultation.appointment_date >= inicio_dia,
            Consultation.appointment_date < fin_dia
        ).order_by(Consultation.appointment_date).all()

        # Trae las citas futuras (desde mañana hasta un máximo de 15 días) con sus detalles.
        citas_futuras = db.session.query(Consultation, Pet, Owner).join(
            Pet, Consultation.pet_id == Pet.id
        ).join(
            Owner, Pet.owner_id == Owner.id
        ).filter(
            Consultation.appointment_date >= fin_dia,
            Consultation.appointment_date <= fin_rango_futuras
        ).order_by(Consultation.appointment_date).all()

    except Exception:
        # Si hay un error en la base de datos, asigna valores por defecto para no romper la vista.
        total_pacientes = 0
        total_duenos = 0
        total_consultas = 0
        consultas_hoy = 0
        citas_hoy = []
        citas_futuras = []

    # Renderiza la plantilla de inicio pasando todos los datos calculados.
    return render_template('inicio.html', hostname=socket.gethostname(),
                           total_pacientes=total_pacientes,
                           total_duenos=total_duenos,
                           total_consultas=total_consultas,
                           consultas_hoy=consultas_hoy,
                           citas_hoy=citas_hoy,
                           citas_futuras=citas_futuras)

@app.route('/registrar', methods=['GET', 'POST'])
@login_required
@permission_required('crear_pacientes')
def registrar_paciente():
    # Si el usuario envía el formulario (POST).
    if request.method == 'POST':
        # Define el número máximo de intentos para el Failover.
        intentos = 3
        # Bucle que intenta guardar los datos varias veces si hay error de conexión.
        for intento in range(intentos):
            try:
                # Crea un nuevo registro de Dueño con los datos del formulario.
                owner = Owner(name=request.form['owner_name'], phone=request.form.get('owner_phone'), email=request.form.get('owner_email'), address=request.form.get('owner_address'))
                # Agrega el dueño a la sesión.
                db.session.add(owner)
                # Fuerza el guardado para obtener el ID del dueño recién creado.
                db.session.flush()

                # Captura la edad y el peso del formulario.
                age = request.form.get('pet_age')
                weight = request.form.get('pet_weight')
                # Crea el registro de la Mascota, convirtiendo edad y peso a números si no están vacíos.
                pet = Pet(owner_id=owner.id, name=request.form['pet_name'],
                          species=request.form.get('pet_species'),
                          breed=request.form.get('pet_breed'),
                          age=int(age) if age else None,
                          weight=float(weight) if weight else None)
                # Agrega la mascota a la sesión.
                db.session.add(pet)
                # Fuerza el guardado para obtener el ID de la mascota.
                db.session.flush()

                # Captura la fecha y hora de la cita.
                appointment_str = request.form.get('appointment_date')
                # Si existe una fecha, la convierte al formato de Python.
                if appointment_str:
                    appointment_dt = datetime.strptime(appointment_str, '%Y-%m-%dT%H:%M')
                else:
                    # Si no hay fecha, usa la fecha y hora actual.
                    appointment_dt = datetime.now()

                # Verifica si ya existe una cita en ese mismo horario exacto.
                cita_existente = Consultation.query.filter_by(appointment_date=appointment_dt).first()
                # Si hay una cita existente, cancela el registro.
                if cita_existente:
                    # Muestra advertencia al usuario.
                    flash('Error: Este horario ya está ocupado.', 'warning')
                    # Deshace los cambios en la base de datos.
                    db.session.rollback()
                    # Redirige de nuevo al formulario.
                    return redirect(url_for('registrar_paciente'))

                # Crea el registro de la consulta/cita con el motivo y diagnóstico.
                consultation = Consultation(pet_id=pet.id, reason=request.form.get('reason'), diagnosis=request.form.get('diagnosis'), appointment_date=appointment_dt)
                # Agrega la consulta a la sesión.
                db.session.add(consultation)
                # Confirma y guarda todos los cambios permanentemente.
                db.session.commit()
                # Muestra mensaje de éxito.
                flash('Paciente y cita registrados exitosamente!', 'success')
                # Redirige a la lista de pacientes.
                return redirect(url_for('pacientes'))
            except Exception as e:
                # Si hay error, deshace todos los cambios de este intento.
                db.session.rollback()
                # Si no es el último intento, espera 1 segundo y reintenta.
                if intento < (intentos - 1):
                    time.sleep(1)
                else:
                    # Si es el último intento, muestra el error al usuario.
                    flash(f'Error al registrar: {str(e)}', 'danger')
    # Renderiza el formulario de registro.
    return render_template('index.html', hostname=socket.gethostname())

@app.route('/paciente/ver/<int:id>')
@login_required
def ver_paciente(id):
    try:
        # Busca la mascota por ID o lanza error 404 si no existe.
        pet = Pet.query.get_or_404(id)
        # Busca al dueño de la mascota.
        owner = Owner.query.get(pet.owner_id)
        # Trae todas las consultas de la mascota ordenadas de la más reciente a la más antigua.
        consultas = Consultation.query.filter_by(pet_id=pet.id).order_by(Consultation.appointment_date.desc()).all()

        bitacora = Bitacora.query.filter_by(pet_id=pet.id).order_by(Bitacora.created_at.desc()).all()
    except Exception:
        # Si hay error en la base de datos, deshace la sesión y redirige.
        db.session.rollback()
        flash('Error de conexión. Recargue la página.', 'danger')
        return redirect(url_for('pacientes'))
    # Renderiza la ficha clínica con todos los datos.
    return render_template('ver_paciente.html', pet=pet, owner=owner,
                           consultas=consultas, bitacora=bitacora,
                           can_add_note=_puede_agregar_nota(),
                           hostname=socket.gethostname())

@app.route('/paciente/editar/<int:id>', methods=['GET', 'POST'])
@login_required
@permission_required('editar_pacientes')
def editar_paciente(id):
    try:
        # Busca la mascota por ID.
        pet = Pet.query.get_or_404(id)
        # Busca al dueño relacionado.
        owner = Owner.query.get(pet.owner_id)
        # Obtiene la última consulta registrada para esa mascota.
        consulta = Consultation.query.filter_by(pet_id=pet.id).order_by(Consultation.appointment_date.desc()).first()
    except Exception:
        # Si hay error, muestra mensaje y redirige a la lista.
        flash('Error de conexión.', 'danger')
        return redirect(url_for('pacientes'))

    # Si el usuario envía el formulario (POST).
    if request.method == 'POST':
        # Define el número máximo de intentos.
        intentos = 3
        # Bucle de reintentos.
        for intento in range(intentos):
            try:
                # Actualiza el nombre del dueño.
                owner.name = request.form['owner_name']
                # Actualiza el teléfono del dueño.
                owner.phone = request.form.get('owner_phone')
                # Actualiza el email del dueño.
                owner.email = request.form.get('owner_email')
                # Actualiza la dirección del dueño.
                owner.address = request.form.get('owner_address')

                # Actualiza el nombre de la mascota.
                pet.name = request.form['pet_name']
                # Actualiza la especie de la mascota.
                pet.species = request.form.get('pet_species')
                # Actualiza la raza de la mascota.
                pet.breed = request.form.get('pet_breed')
                # Captura la edad y el peso del formulario.
                age = request.form.get('pet_age')
                weight = request.form.get('pet_weight')
                # Convierte a entero si existe, sino guarda None.
                pet.age = int(age) if age else None
                # Convierte a flotante si existe, sino guarda None.
                pet.weight = float(weight) if weight else None

                # Si existe una consulta previa, actualiza sus datos.
                if consulta:
                    # Captura la nueva fecha de la cita.
                    appointment_str = request.form.get('appointment_date')
                    # Si hay fecha, la convierte.
                    if appointment_str:
                        nueva_fecha = datetime.strptime(appointment_str, '%Y-%m-%dT%H:%M')
                        
                        # Verifica si ya existe otra cita en ese mismo horario (excluyendo la cita actual).
                        conflicto = Consultation.query.filter(
                            Consultation.appointment_date == nueva_fecha,
                            Consultation.id != consulta.id
                        ).first()
                        
                        # Si hay conflicto, cancela la actualización y avisa al usuario.
                        if conflicto:
                            flash('Error: Este horario ya está ocupado por otro paciente.', 'warning')
                            db.session.rollback()
                            return redirect(url_for('editar_paciente', id=id))
                            
                        # Si no hay conflicto, actualiza la fecha.
                        consulta.appointment_date = nueva_fecha
                        
                    # Actualiza el motivo.
                    consulta.reason = request.form.get('reason')
                    # Actualiza el diagnóstico/notas.
                    consulta.diagnosis = request.form.get('diagnosis')

                # Confirma todos los cambios.
                db.session.commit()
                # Muestra mensaje de éxito.
                flash('Actualizado correctamente!', 'success')
                # Redirige a la lista.
                return redirect(url_for('pacientes'))
            except Exception as e:
                # Deshace los cambios si hay error.
                db.session.rollback()
                # Si no es el último intento, espera y reintenta.
                if intento < (intentos - 1):
                    time.sleep(1)
                else:
                    # Muestra el error en el último intento.
                    flash(f'Error al actualizar: {str(e)}', 'danger')

    # Renderiza el formulario de edición con los datos actuales.
    return render_template('editar_paciente.html', pet=pet, owner=owner,
                           consulta=consulta, hostname=socket.gethostname())

# --- RUTA: ELIMINAR PACIENTE ---
@app.route('/paciente/eliminar/<int:id>')
@login_required
@permission_required('eliminar_pacientes')
def eliminar_paciente(id):
    # Define el número máximo de intentos.
    intentos = 3
    # Bucle de reintentos.
    for intento in range(intentos):
        try:
            # Busca la mascota a eliminar.
            pet = Pet.query.get_or_404(id)
            # Guarda el ID del dueño antes de borrar la mascota.
            owner_id = pet.owner_id

            # Elimina primero las consultas asociadas para no violar la llave foránea.
            Consultation.query.filter_by(pet_id=pet.id).delete()
            # Elimina también las notas de bitácora asociadas.
            Bitacora.query.filter_by(pet_id=pet.id).delete()

            # Elimina el registro de la mascota.
            db.session.delete(pet)
            # Fuerza el guardado para verificar que se eliminó.
            db.session.flush()

            # Cuenta cuántas mascotas le quedan al dueño.
            otras_mascotas = Pet.query.filter_by(owner_id=owner_id).count()
            # Si ya no tiene mascotas, elimina también al dueño.
            if otras_mascotas == 0:
                # Busca al dueño.
                owner = Owner.query.get(owner_id)
                # Si existe, lo elimina.
                if owner:
                    db.session.delete(owner)

            # Confirma todos los cambios.
            db.session.commit()
            # Muestra mensaje de éxito.
            flash('Paciente eliminado.', 'success')
            # Redirige a la lista.
            return redirect(url_for('pacientes'))
        except Exception as e:
            # Deshace los cambios si hay error.
            db.session.rollback()
            # Si no es el último intento, espera y reintenta.
            if intento < (intentos - 1):
                time.sleep(1)
            else:
                # Muestra el error en el último intento.
                flash(f'Error al eliminar: {str(e)}', 'danger')
    # Redirige a la lista al terminar.
    return redirect(url_for('pacientes'))

            # Confirma y guarda.
            db.session.commit()
            # Muestra mensaje de éxito.
            flash('Usuario creado exitosamente!', 'success')
        except Exception:
            # Deshace si hay error.
            db.session.rollback()
            # Muestra mensaje de error.
            flash('Error al crear el usuario.', 'danger')

    try:
        # Trae todos los usuarios para la tabla.
        lista_users = User.query.all()
        # Trae todos los roles para el select del formulario.
        lista_roles = Role.query.all()
    except Exception:
        # Si hay error, listas vacías.
        lista_users = []
        lista_roles = []

    return render_template('usuarios.html', usuarios=lista_users, roles=lista_roles, hostname=socket.gethostname())

@app.route('/usuario/editar/<int:id>', methods=['GET', 'POST'])
@login_required
@permission_required('gestionar_usuarios')
def editar_usuario(id):
    try:
        # Busca el usuario por ID.
        user = User.query.get_or_404(id)
        # Trae todos los roles para el select.
        lista_roles = Role.query.all()
    except Exception:
        # Si hay error, muestra mensaje y redirige.
        flash('Error de conexión.', 'danger')
        return redirect(url_for('usuarios'))

    # Si el usuario envía el formulario (POST).
    if request.method == 'POST':
        try:
            # Captura el nuevo nombre de usuario.
            nuevo_username = request.form['username']
            # Verifica si el nombre ya está en uso por otro usuario.
            usuario_existente = User.query.filter_by(username=nuevo_username).first()
            # Si existe y no es el mismo usuario, cancela el cambio.
            if usuario_existente and usuario_existente.id != user.id:
                flash('El nombre de usuario ya está en uso.', 'danger')
                return redirect(url_for('editar_usuario', id=user.id))

            # Actualiza el nombre de usuario.
            user.username = nuevo_username
            # Actualiza el rol.
            user.role = request.form['role']

            # Captura la nueva contraseña (si se escribió).
            new_password = request.form.get('password')
            # Si hay nueva contraseña, la encripta y actualiza.
            if new_password:
                user.password_hash = generate_password_hash(new_password)

            # Confirma los cambios.
            db.session.commit()
            # Muestra mensaje de éxito.
            flash('Usuario actualizado!', 'success')
            # Redirige a la lista de usuarios.
            return redirect(url_for('usuarios'))
        except Exception as e:
            # Deshace si hay error.
            db.session.rollback()
            # Muestra el error.
            flash(f'Error al actualizar: {str(e)}', 'danger')

    # Renderiza el formulario de edición.
    return render_template('editar_usuario.html', user=user, roles=lista_roles, hostname=socket.gethostname())

            db.session.add(nuevo_role)
            # Confirma y guarda.
            db.session.commit()
            # Muestra mensaje de éxito.
            flash('Rol creado exitosamente!', 'success')
            # Redirige a la misma página.
            return redirect(url_for('roles'))
        except Exception:
            # Deshace si hay error.
            db.session.rollback()
            # Muestra error (puede ser rol duplicado).
            flash('Error al crear el rol.', 'danger')

    # Trae todos los roles existentes para la tabla.
    lista_roles = Role.query.all()
    # Renderiza la plantilla de roles.
    return render_template('roles.html', roles=lista_roles, hostname=socket.gethostname())
