    <!-- Estilos personalizados para centrar el login y dar el fondo con gradiente. -->

    <div class="card login-card shadow-lg border-0 rounded-4">
        <div class="card-body p-5">
            <div class="text-center mb-4">
                <!-- Ícono de pulso cardíaco y título. -->
                <i class="bi bi-heart-pulse fs-1 text-primary"></i>
                <h3 class="fw-bold mt-2">VetSystem</h3>
            </div>
            
            <!-- Bloque de Jinja2 para mostrar mensajes flash (errores de contraseña). -->
            {% with messages = get_flashed_messages(with_categories=true) %}
              {% if messages %}
                {% for category, message in messages %}
                  <!-- Alerta de Bootstrap con color según la categoría (danger = rojo). -->
                  <div class="alert alert-{{ category }} p-2 text-center">{{ message }}</div>
                {% endfor %}
              {% endif %}
            {% endwith %}
            
            <!-- Formulario que envía los datos a la ruta /login por método POST. -->
            <form method="POST" action="/login">
                <div class="mb-3">
                    <label class="form-label">Usuario</label>
                    <!-- El atributo name="username" es vital para que Python lo lea. -->
                    <input type="text" class="form-control" name="username" required>
                </div>
                <div class="mb-3">
                    <label class="form-label">Contraseña</label>
                    <!-- El tipo password oculta los caracteres con puntos. -->
                    <input type="password" class="form-control" name="password" required>
                </div>
                <button type="submit" class="btn btn-primary w-100">Ingresar</button>
            </form>
        </div>
    </div>
<!-- Script de Bootstrap necesario para que funcionen las alertas. -->
